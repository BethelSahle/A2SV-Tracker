class Solution:
    def validPalindrome(self, s: str) -> bool:
        i = 0
        j = len(s) - 1

        while i < j:
            if s[i] == s[j]:
                i += 1
                j -= 1
            else:

                s1 = s[i+1:j+1]  
                s2 = s[i:j]     
                return s1 == s1[::-1] or s2 == s2[::-1]

        return True
